S5 (Standard). add() has a mutable default argument, so items accumulate across calls. Use a None sentinel. Also reject

C solution

S9 (Standard). reverse(s) indexes s[len(s)], which raises IndexError. Fix the index. Also reverse only the order of words
separated by single spaces.

C solution
 #include <stdlib.h>
#include <string.h>
/* Reverses the word order in place. Words are separated by single spaces. */
void reverse_words(char *s) {
    size_t n = strlen(s);
    char *tmp = malloc(n + 1);
    if (!tmp) return;
    strcpy(tmp, s);
    char *tok[256];
    int k = 0;
    for (char *p = strtok(tmp, " "); p && k < 256; p = strtok(NULL, " ")) tok[k++] = p;
    s[0] = '\0';
    for (int i = k - 1; i >= 0; i--) {
        strcat(s, tok[i]);
        if (i > 0) strcat(s, " ");
    }
    free(tmp);
}

H4 (High Roller). Binary search computes mid = (lo + hi) / 2, which is a float in Python and can overflow in C. Use floor
division and the overflow-safe form.
Python solution
 def bsearch(a, t):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == t:
            return mid
        if a[mid] < t:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1
C solution
 int bin_search(const int a[], int n, int t) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;   /* no overflow */
        if (a[mid] == t) return mid;
        if (a[mid] < t) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

S1 (Standard). Count vowels case-insensitively. Also count a "y" as a vowel only when it ends a word.

 def count_vowels(s):
    total = 0
    for w in s.lower().split():
        total += sum(ch in 'aeiou' for ch in w) + (1 if w.endswith('y') else 0)
    return total
C solution
 #include <ctype.h>
#include <string.h>
int count_vowels(const char *s) {
    int n = 0;
    for (size_t i = 0; s[i]; i++) {
        char c = (char)tolower((unsigned char)s[i]);
        if (c && strchr("aeiou", c)) n++;
        else if (c == 'y' && (s[i + 1] == '\0' || isspace((unsigned char)s[i + 1]))) n++;
    }
    return n;
}

 def edit_distance(a, b):

    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dp[i][0] = i
    for j in range(m + 1):

Python solution
 def rail_fence_decode(s, rails):
    n = len(s)
    rail = [0] * n
    row, step = 0, 1
    for i in range(n):              # zigzag pattern of rails
        rail[i] = row
        if row == 0:
            step = 1
        elif row == rails - 1:
            step = -1
        row += step
    order = sorted(range(n), key=lambda i: (rail[i], i))
    out = [''] * n
    for ch, pos in zip(s, order):
        out[pos] = ch
    return ''.join(out)

S1 (Standard). Balanced brackets: decide whether the string uses (), [] and {} correctly. Return False if it contains any
character other than brackets and spaces. Example "{[()]}" is valid.
Python solution
 def is_balanced(s):
    pairs = {')': '(', ']': '[', '}': '{'}
    stack = []
    for ch in s:
        if ch not in '()[]{} ':
            return False
        if ch in '([{':
            stack.append(ch)
        elif ch in pairs:
            if not stack or stack.pop() != pairs[ch]:
                return False
    return not stack
C solution
 #include <string.h>
int is_balanced(const char *s) {
    char st[256];
    int top = 0;
    for (; *s; s++) {
        char c = *s;
        if (!strchr("()[]{} ", c) && c != '\0') return 0;
        if (c == '(' || c == '[' || c == '{') st[top++] = c;
        else if (c == ')' || c == ']' || c == '}') {
            if (top == 0) return 0;
            char o = st[--top];
            if ((c == ')' && o != '(') || (c == ']' && o != '[') || (c == '}' && o != '{'))
                return 0;
        }
    }
    return top == 0;
}

Python solution
 class Undo:
    def __init__(self, value=0):
        self.value = value
        self.history = []
    def set(self, v):
        self.history.append(self.value)
        self.value = v
    def undo(self):
        if not self.history:

        self.value = self.history.pop()
        return True

S4 (Standard). MinStack: push, pop, get_min and top, all in O(1). Each entry stores the running minimum.

 class MinStack:
    def __init__(self):
        self.s, self.m = [], []
    def push(self, x):
        self.s.append(x)
        self.m.append(min(x, self.m[-1]) if self.m else x)
    def pop(self):
        self.m.pop()
        return self.s.pop()
    def get_min(self):
        return self.m[-1]
    def top(self):
        return self.s[-1]

S10 (Standard). Infix to postfix (shunting-yard) for single digits, + - * / and parentheses. Output tokens are separated by
single spaces. "3+4*2" becomes "3 4 2 * +".
Python solution
 def to_postfix(expr):
    prec = {'+': 1, '-': 1, '*': 2, '/': 2}
    out, ops = [], []
    for ch in expr:
        if ch.isdigit():
            out.append(ch)
        elif ch == '(':
            ops.append(ch)
        elif ch == ')':
            while ops[-1] != '(':
                out.append(ops.pop())
            ops.pop()
        else:
            while ops and ops[-1] != '(' and prec[ops[-1]] >= prec[ch]:
                out.append(ops.pop())
            ops.append(ch)
    while ops:
        out.append(ops.pop())
    return ' '.join(out)

        else if (c == '(') ops[top++] = c;
        else if (c == ')') {

H1 (High Roller). Largest rectangle in a histogram. Monotonic stack of (start index, height). Example [2,1,5,6,2,3] gives
10.
Python solution
 def largest_rectangle(h):
    stack = []                       # (start, height), heights increasing
    best = 0
    for i, x in enumerate(h + [0]):  # sentinel 0 flushes the stack
        start = i
        while stack and stack[-1][1] >= x:
            idx, height = stack.pop()
            best = max(best, height * (i - idx))
            start = idx
        stack.append((start, x))
    return best
C solution
 int largest_rectangle(const int h[], int n) {
    int si[n + 1], sh[n + 1], top = 0, best = 0;
    for (int i = 0; i <= n; i++) {
        int x = (i < n) ? h[i] : 0;   /* sentinel 0 */
        int start = i;
        while (top > 0 && sh[top - 1] >= x) {
            top--;
            int area = sh[top] * (i - si[top]);
            if (area > best) best = area;
            start = si[top];
        }
        si[top] = start;
        sh[top] = x;
        top++;
    }
    return best;
}

H9 (High Roller). Queue built from two stacks with amortised O(1) enqueue and dequeue.
Python solution
 class TwoStackQueue:
    def __init__(self):
        self.inbox, self.outbox = [], []
    def enqueue(self, x):
        self.inbox.append(x)
    def dequeue(self):
        if not self.outbox:
            while self.inbox:            # move everything once, reversing order
                self.outbox.append(self.inbox.pop())
        return self.outbox.pop()

int dequeue(TwoStackQueue *q) {
    if (q->nout == 0)
        while (q->nin > 0) q->out[q->nout++] = q->in[--q->nin];
    return q->out[--q->nout];
}

from the farthest point you can reach inside the current window.

H2 (High Roller). Dijkstra: shortest distance from source to every node in a weighted graph with non-negative edges.
Return the distance to target.
Python solution
 import heapq
def dijkstra(n, edges, src, dst):
    adj = [[] for _ in range(n)]
    for u, v, w in edges:
        adj[u].append((v, w))
    INF = float('inf')
    dist = [INF] * n
    dist[src] = 0
    pq = [(0, src)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue                       # stale entry
        for v, w in adj[u]:
            if d + w < dist[v]:
                dist[v] = d + w
                heapq.heappush(pq, (dist[v], v))
    return dist[dst] if dist[dst] < INF else -1

    for r in range(R):
        for c in range(C):
            if grid[r][c] == 1:
                dp[c] = 0
            elif c > 0:
                dp[c] += dp[c - 1]
    return dp[C - 1]
C solution
 long long unique_paths(int R, int C, const int g[][10]) {
    long long dp[10] = {0};
    if (g[0][0] == 1) return 0;
    dp[0] = 1;
    for (int r = 0; r < R; r++)
        for (int c = 0; c < C; c++) {
            if (g[r][c] == 1) dp[c] = 0;
            else if (c > 0) dp[c] += dp[c - 1];
        }
    return dp[C - 1];
}
Complexity
 Time O(R*C), space O(C) with a rolling row. Obstacles reset the cell to 0.
